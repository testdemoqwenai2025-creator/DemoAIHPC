   220→// ============ QUANTUM BACKGROUND COMPONENT ============
   221→interface QuantumBackgroundProps {
   222→  darkMode: boolean
   223→  enabled: boolean
   224→}
   225→
   226→function QuantumBackground({ darkMode, enabled }: QuantumBackgroundProps) {
   227→  const canvasRef = useRef<HTMLCanvasElement>(null)
   228→  
   229→  useEffect(() => {
   230→    if (!enabled) return
   231→    
   232→    const canvas = canvasRef.current
   233→    if (!canvas) return
   234→    
   235→    const ctx = canvas.getContext('2d')
   236→    if (!ctx) return
   237→    
   238→    let animationId: number
   239→    const particles: Array<{
   240→      x: number; y: number; vx: number; vy: number; 
   241→      radius: number; phase: number; frequency: number; amplitude: number
   242→    }> = []
   243→    
   244→    const resizeCanvas = () => {
   245→      canvas.width = window.innerWidth
   246→      canvas.height = window.innerHeight
   247→    }
   248→    
   249→    resizeCanvas()
   250→    window.addEventListener('resize', resizeCanvas)
   251→    
   252→    for (let i = 0; i < 80; i++) {
   253→      particles.push({
   254→        x: Math.random() * canvas.width,
   255→        y: Math.random() * canvas.height,
   256→        vx: (Math.random() - 0.5) * 0.5,
   257→        vy: (Math.random() - 0.5) * 0.5,
   258→        radius: Math.random() * 2 + 1,
   259→        phase: Math.random() * Math.PI * 2,
   260→        frequency: 0.01 + Math.random() * 0.02,
   261→        amplitude: 20 + Math.random() * 40
   262→      })
   263→    }
   264→    
   265→    const animate = (time: number) => {
   266→      ctx.fillStyle = darkMode ? 'rgba(10, 10, 30, 0.15)' : 'rgba(248, 250, 252, 0.15)'
   267→      ctx.fillRect(0, 0, canvas.width, canvas.height)
   268→      
   269→      particles.forEach((p, i) => {
   270→        p.x += p.vx + Math.sin(time * p.frequency + p.phase) * 0.3
   271→        p.y += p.vy + Math.cos(time * p.frequency + p.phase) * 0.3
   272→        
   273→        if (p.x < 0) p.x = canvas.width
   274→        if (p.x > canvas.width) p.x = 0
   275→        if (p.y < 0) p.y = canvas.height
   276→        if (p.y > canvas.height) p.y = 0
   277→        
   278→        const gradient = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.radius * 3)
   279→        const hue = (time * 0.02 + i * 5) % 360
   280→        gradient.addColorStop(0, `hsla(${hue}, 80%, 60%, 0.8)`)
   281→        gradient.addColorStop(0.5, `hsla(${hue}, 70%, 50%, 0.3)`)
   282→        gradient.addColorStop(1, 'transparent')
   283→        
   284→        ctx.beginPath()
   285→        ctx.arc(p.x, p.y, p.radius * 3, 0, Math.PI * 2)
   286→        ctx.fillStyle = gradient
   287→        ctx.fill()
   288→        
   289→        particles.slice(i + 1).forEach(p2 => {
   290→          const dx = p.x - p2.x
   291→          const dy = p.y - p2.y
   292→          const dist = Math.sqrt(dx * dx + dy * dy)
   293→          
   294→          if (dist < 150) {
   295→            ctx.beginPath()
   296→            ctx.moveTo(p.x, p.y)
   297→            ctx.lineTo(p2.x, p2.y)
   298→            ctx.strokeStyle = `hsla(${hue}, 60%, 50%, ${0.15 * (1 - dist / 150)})`
   299→            ctx.lineWidth = 0.5
   300→            ctx.stroke()
   301→          }
   302→        })
   303→      })
   304→      
   305→      animationId = requestAnimationFrame(animate)
   306→    }
   307→    
   308→    animate(0)
   309→    
   310→    return () => {
   311→      cancelAnimationFrame(animationId)
   312→      window.removeEventListener('resize', resizeCanvas)
   313→    }
   314→  }, [enabled, darkMode])
   315→

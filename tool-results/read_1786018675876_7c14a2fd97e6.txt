   118→// ============ ANIMATED BACKGROUND ============
   119→function ParticleField() {
   120→  const canvasRef = useRef<HTMLCanvasElement>(null)
   121→  
   122→  useEffect(() => {
   123→    const canvas = canvasRef.current
   124→    if (!canvas) return
   125→    
   126→    const ctx = canvas.getContext('2d')
   127→    if (!ctx) return
   128→    
   129→    let animationId: number
   130→    const particles: Array<{
   131→      x: number; y: number; vx: number; vy: number; 
   132→      size: number; opacity: number; color: string
   133→    }> = []
   134→    
   135→    const colors = ['#14b8a6', '#06b6d4', '#3b82f6', '#8b5cf6', '#ec4899']
   136→    
   137→    const resize = () => {
   138→      canvas.width = window.innerWidth
   139→      canvas.height = window.innerHeight
   140→    }
   141→    
   142→    const createParticle = () => ({
   143→      x: Math.random() * canvas.width,
   144→      y: Math.random() * canvas.height,
   145→      vx: (Math.random() - 0.5) * 0.5,
   146→      vy: (Math.random() - 0.5) * 0.5,
   147→      size: Math.random() * 2 + 1,
   148→      opacity: Math.random() * 0.5 + 0.2,
   149→      color: colors[Math.floor(Math.random() * colors.length)]
   150→    })
   151→    
   152→    const init = () => {
   153→      resize()
   154→      for (let i = 0; i < 80; i++) {
   155→        particles.push(createParticle())
   156→      }
   157→    }
   158→    
   159→    const animate = () => {
   160→      ctx.clearRect(0, 0, canvas.width, canvas.height)
   161→      
   162→      particles.forEach((p, i) => {
   163→        p.x += p.vx
   164→        p.y += p.vy
   165→        
   166→        if (p.x < 0 || p.x > canvas.width) p.vx *= -1
   167→        if (p.y < 0 || p.y > canvas.height) p.vy *= -1
   168→        
   169→        ctx.beginPath()
   170→        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2)
   171→        ctx.fillStyle = p.color
   172→        ctx.globalAlpha = p.opacity
   173→        ctx.fill()
   174→        
   175→        // Connect nearby particles
   176→        particles.slice(i + 1).forEach(p2 => {
   177→          const dx = p.x - p2.x
   178→          const dy = p.y - p2.y
   179→          const dist = Math.sqrt(dx * dx + dy * dy)
   180→          
   181→          if (dist < 150) {
   182→            ctx.beginPath()
   183→            ctx.moveTo(p.x, p.y)
   184→            ctx.lineTo(p2.x, p2.y)
   185→            ctx.strokeStyle = p.color
   186→            ctx.globalAlpha = (1 - dist / 150) * 0.15
   187→            ctx.lineWidth = 0.5
   188→            ctx.stroke()
   189→          }
   190→        })
   191→      })
   192→      
   193→      ctx.globalAlpha = 1
   194→      animationId = requestAnimationFrame(animate)
   195→    }
   196→    
   197→    init()
   198→    animate()
   199→    
   200→    window.addEventListener('resize', resize)
   201→    
   202→    return () => {
   203→      cancelAnimationFrame(animationId)
   204→      window.removeEventListener('resize', resize)
   205→    }
   206→  }, [])
   207→  
   208→  return (
   209→    <canvas 
   210→      ref={canvasRef} 
   211→      className="absolute inset-0 pointer-events-none"
   212→      style={{ background: 'linear-gradient(135deg, #0c1929 0%, #0f172a 50%, #1e1030 100%)' }}
   213→    />
   214→  )
   215→}

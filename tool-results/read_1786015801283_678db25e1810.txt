   426→// ============ ANIMATED BACKGROUND ============
   427→function ArchitectureBackground() {
   428→  const canvasRef = useRef<HTMLCanvasElement>(null)
   429→  
   430→  useEffect(() => {
   431→    const canvas = canvasRef.current
   432→    if (!canvas) return
   433→    
   434→    const ctx = canvas.getContext('2d')
   435→    if (!ctx) return
   436→    
   437→    let animationId: number
   438→    let time = 0
   439→    
   440→    const nodes: Array<{x: number; y: number; vx: number; vy: number; size: number; connections: number}> = []
   441→    
   442→    const resize = () => {
   443→      canvas.width = window.innerWidth
   444→      canvas.height = window.innerHeight
   445→    }
   446→    
   447→    const init = () => {
   448→      resize()
   449→      for (let i = 0; i < 60; i++) {
   450→        nodes.push({
   451→          x: Math.random() * canvas.width,
   452→          y: Math.random() * canvas.height,
   453→          vx: (Math.random() - 0.5) * 0.3,
   454→          vy: (Math.random() - 0.5) * 0.3,
   455→          size: Math.random() * 2 + 1,
   456→          connections: 0
   457→        })
   458→      }
   459→    }
   460→    
   461→    const animate = () => {
   462→      ctx.fillStyle = 'rgba(2, 6, 23, 0.15)'
   463→      ctx.fillRect(0, 0, canvas.width, canvas.height)
   464→      
   465→      time += 0.01
   466→      
   467→      // Draw grid
   468→      ctx.strokeStyle = 'rgba(139, 92, 246, 0.03)'
   469→      ctx.lineWidth = 1
   470→      const gridSize = 50
   471→      for (let x = 0; x < canvas.width; x += gridSize) {
   472→        for (let y = 0; y < canvas.height; y += gridSize) {
   473→          ctx.strokeRect(x, y, gridSize, gridSize)
   474→        }
   475→      }
   476→      
   477→      // Update and draw nodes
   478→      nodes.forEach((node, i) => {
   479→        node.x += node.vx
   480→        node.y += node.vy
   481→        
   482→        if (node.x < 0 || node.x > canvas.width) node.vx *= -1
   483→        if (node.y < 0 || node.y > canvas.height) node.vy *= -1
   484→        
   485→        // Node glow
   486→        const gradient = ctx.createRadialGradient(node.x, node.y, 0, node.x, node.y, node.size * 8)
   487→        gradient.addColorStop(0, `rgba(6, 182, 212, 0.3)`)
   488→        gradient.addColorStop(1, 'transparent')
   489→        ctx.fillStyle = gradient
   490→        ctx.beginPath()
   491→        ctx.arc(node.x, node.y, node.size * 8, 0, Math.PI * 2)
   492→        ctx.fill()
   493→        
   494→        // Node core
   495→        ctx.fillStyle = `hsl(${180 + i * 3}, 70%, 60%)`
   496→        ctx.beginPath()
   497→        ctx.arc(node.x, node.y, node.size, 0, Math.PI * 2)
   498→        ctx.fill()
   499→        
   500→        // Connections
   501→        nodes.slice(i + 1).forEach((other) => {
   502→          const dx = node.x - other.x
   503→          const dy = node.y - other.y
   504→          const dist = Math.sqrt(dx * dx + dy * dy)
   505→          
   506→          if (dist < 150) {
   507→            ctx.beginPath()
   508→            ctx.moveTo(node.x, node.y)
   509→            
   510→            // Curved connection
   511→            const midX = (node.x + other.x) / 2 + Math.sin(time + i) * 20
   512→            const midY = (node.y + other.y) / 2 + Math.cos(time + i) * 20
   513→            ctx.quadraticCurveTo(midX, midY, other.x, other.y)
   514→            
   515→            ctx.strokeStyle = `rgba(139, 92, 246, ${(1 - dist / 150) * 0.2})`
   516→            ctx.lineWidth = 0.5
   517→            ctx.stroke()
   518→          }
   519→        })
   520→      })
   521→      
   522→      animationId = requestAnimationFrame(animate)
   523→    }
   524→    
   525→    init()
   526→    animate()
   527→    
   528→    window.addEventListener('resize', resize)
   529→    return () => {
   530→      cancelAnimationFrame(animationId)
   531→      window.removeEventListener('resize', resize)
   532→    }
   533→  }, [])
   534→  
   535→  return (
   536→    <canvas 
   537→      ref={canvasRef} 
   538→      className="fixed inset-0 pointer-events-none z-0"
   539→      style={{ background: 'linear-gradient(135deg, #020617 0%, #0f172a 30%, #1e1b4b 60%, #0c0a1e 100%)' }}
   540→    />
   541→  )
   542→}

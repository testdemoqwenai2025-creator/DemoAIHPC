   226→   226→function QuantumBackground({ darkMode, enabled }: QuantumBackgroundProps) {
   227→   227→  const canvasRef = useRef<HTMLCanvasElement>(null)
   228→   228→  
   229→   229→  useEffect(() => {
   230→   230→    if (!enabled) return
   231→   231→    
   232→   232→    const canvas = canvasRef.current
   233→   233→    if (!canvas) return
   234→   234→    
   235→   235→    const ctx = canvas.getContext('2d')
   236→   236→    if (!ctx) return
   237→   237→    
   238→   238→    let animationId: number
   239→   239→    const particles: Array<{
   240→   240→      x: number; y: number; vx: number; vy: number; 
   241→   241→      radius: number; phase: number; frequency: number; amplitude: number
   242→   242→    }> = []
   243→   243→    
   244→   244→    const resizeCanvas = () => {
   245→   245→      canvas.width = window.innerWidth
   246→   246→      canvas.height = window.innerHeight
   247→   247→    }
   248→   248→    
   249→   249→    resizeCanvas()
   250→   250→    window.addEventListener('resize', resizeCanvas)
   251→   251→    
   252→   252→    for (let i = 0; i < 80; i++) {
   253→   253→      particles.push({
   254→   254→        x: Math.random() * canvas.width,
   255→   255→        y: Math.random() * canvas.height,
   256→   256→        vx: (Math.random() - 0.5) * 0.5,
   257→   257→        vy: (Math.random() - 0.5) * 0.5,
   258→   258→        radius: Math.random() * 2 + 1,
   259→   259→        phase: Math.random() * Math.PI * 2,
   260→   260→        frequency: 0.01 + Math.random() * 0.02,
   261→   261→        amplitude: 20 + Math.random() * 40
   262→   262→      })
   263→   263→    }
   264→   264→    
   265→   265→    const animate = (time: number) => {

   312→   312→      window.removeEventListener('resize', resizeCanvas)
   313→   313→    }
   314→   314→  }, [enabled, darkMode])

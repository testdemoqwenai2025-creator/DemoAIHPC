L = (1/n) · Σ |y_i − ŷ_i|
# L = (1/n) · Σ |y_i − ŷ_i|
# |·| → np.abs()   (1/n) → np.mean()
import numpy as np

def mae(y_true, y_pred):
    # L = mean of absolute differences
    return np.mean(np.abs(y_true - y_pred))

# Example
y_true = np.array([1.0, 2.0, 3.0])
y_pred = np.array([1.5, 1.5, 3.5])
print(mae(y_true, y_pred))  # 0.5

x̂ = x / √(mean(x²) + ε) y = γ·x̂
# x̂ = x / √(mean(x²) + ε)  then  y = γ·x̂
# No mean subtraction, no β shift. Simpler than LayerNorm.
import numpy as np

def rmsnorm(x, gamma=1.0, eps=1e-6):
    rms = np.sqrt(np.mean(x**2, axis=-1, keepdims=True) + eps)
    return gamma * x / rms

# Forward (noising) — closed form for any t:
#   x_t = sqrt(α_bar_t) * x_0 + sqrt(1 - α_bar_t) * ε,  ε ~ N(0, I)
#   where α_bar_t = Π α_s  (cumulative product of noise schedule)
# Reverse — train network ε_θ to predict the noise added:
#   Loss = E[ || ε - ε_θ(x_t, t) ||² ]
# Sampling: x_{t-1} = (1/√α_t)(x_t - (1-α_t)/√(1-α_bar_t) ε_θ(x_t, t)) + σ_t z
import numpy as np

class Diffusion:
    def __init__(self, num_steps=1000, beta_start=1e-4, beta_end=0.02):
        # Linear noise schedule (DDPM default)
        self.betas = np.linspace(beta_start, beta_end, num_steps)
        self.alphas = 1 - self.betas
        self.alpha_bars = np.cumprod(self.alphas)
        self.num_steps = num_steps

    def forward_noise(self, x_0, t):
        # Closed-form: x_t = √ᾱ_t · x_0 + √(1-ᾱ_t) · ε
        eps = np.random.randn(*x_0.shape)
        alpha_bar = self.alpha_bars[t]
        return np.sqrt(alpha_bar) * x_0 + np.sqrt(1 - alpha_bar) * eps, eps

    def reverse_step(self, x_t, t, eps_pred):
        beta_t = self.betas[t]
        alpha_t = self.alphas[t]
        alpha_bar_t = self.alpha_bars[t]
        mean = (1 / np.sqrt(alpha_t)) * (x_t - (1 - alpha_t) / np.sqrt(1 - alpha_bar_t) * eps_pred)
        if t > 0:
            return mean + np.sqrt(beta_t) * np.random.randn(*x_t.shape)
        return mean

# Example: 5-step toy diffusion
np.random.seed(0)
diff = Diffusion(num_steps=5)
x_0 = np.array([0.5, -0.3, 0.8])
x_t, eps = diff.forward_noise(x_0, t=4)
print(f'Noised x_t: {x_t}')
print(f'True noise ε: {eps}')

🧠 Deep thoughts — when (and when not) to reach for this architecture

USE WHEN:high-quality image generation (SOTA — Stable Diffusion, DALL-E 3, Midjourney v6, Flux all use diffusion), audio synthesis (MusicGen, AudioLDM), video generation (Sora, Runway Gen-3), 3D model generation, molecule/drug design, you need diversity + quality (Diffusion gives both — GAN gives quality but not diversity, VAE gives diversity but not quality).
AVOID WHEN:real-time generation is required (50-1000 reverse steps are slow — use distillation to 1-4 steps for production, or use GAN), strict memory budget (each step stores activations — Latent Diffusion reduces this by running in VAE latent space), small datasets (Diffusion benefits enormously from scale).
WHY:Diffusion Models won because they solved the GAN training stability problem while keeping quality. Instead of an adversarial minimax game, they use a simple denoising objective: predict the noise added to the input. This objective is stable, parallel-friendly, and gives diverse samples because each reverse step adds fresh noise. The cost is slow sampling (many steps), which is being actively addressed by distillation (Consistency Models) and fast samplers (DPM-Solver, DDIM).

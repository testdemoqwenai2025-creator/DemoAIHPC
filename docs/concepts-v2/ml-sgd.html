θₜ₊₁ = θₜ − η · ∂L/∂θ
# θ_{t+1} = θ_t − η · ∂L/∂θ
# θ → params (list of arrays)
# η → lr (learning rate, e.g. 0.01)
# ∂L/∂θ → grads (same shape as params)
import numpy as np

def sgd_step(params, grads, lr=0.01):
    # In-place update: params -= lr * grads
    for i in range(len(params)):
        params[i] -= lr * grads[i]
    return params

# Example: minimise (x-3)^2 + (y+2)^2, minimum at (3, -2)
params = [np.array(0.0), np.array(0.0)]
for step in range(100):
    grads = [2 * (params[0] - 3), 2 * (params[1] + 2)]
    params = sgd_step(params, grads, lr=0.1)
print(params)  # ~[3.0, -2.0]

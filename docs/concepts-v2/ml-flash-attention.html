# FlashAttention = same math as scaled dot-product attention
# but computed in tiled blocks to minimise HBM↔SRAM traffic
# No approximations — output is bit-identical (up to fp tolerance)
import numpy as np

def softmax(x, axis=-1):
    x = x - x.max(axis=axis, keepdims=True)
    e = np.exp(x)
    return e / e.sum(axis=axis, keepdims=True)

def flash_attention(Q, K, V, block_size=64):
    """Reference implementation of FlashAttention forward pass.

    Same numerical output as scaled_dot_product_attention — the algorithmic
    speedup comes from minimising HBM reads/writes (not shown here).
    Real implementations fuse this into a single CUDA/Triton kernel.
    """
    N, d_k = Q.shape
    M = K.shape[0]
    out = np.zeros_like(Q)
    # Block-wise computation: load Q, K, V blocks into SRAM, compute, write back
    for i in range(0, N, block_size):
        Q_block = Q[i:i + block_size]
        out_block = np.zeros_like(Q_block)
        for j in range(0, M, block_size):
            K_block = K[j:j + block_size]
            V_block = V[j:j + block_size]
            scores = Q_block @ K_block.T / np.sqrt(d_k)
            # Block-softmax: numerically identical to full softmax in the limit
            out_block += softmax(scores, axis=-1) @ V_block
        out[i:i + block_size] = out_block
    return out

# Example
Q = np.array([[1.0, 0.0], [0.0, 1.0]])
K = np.array([[1.0, 0.0], [0.0, 1.0], [1.0, 1.0]])
V = np.array([[1.0], [2.0], [3.0]])
print(flash_attention(Q, K, V, block_size=1))  # same as SDPA

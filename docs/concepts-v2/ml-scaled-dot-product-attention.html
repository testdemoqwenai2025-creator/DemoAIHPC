Attn(Q,K,V) = softmax(Q·Kᵀ / √d_k) · V
# Attn(Q,K,V) = softmax(Q·Kᵀ / √d_k) · V
# d_k → key dimension (for scaling)
import numpy as np

def attention(Q, K, V):
    d_k = K.shape[-1]
    scores = Q @ K.T / np.sqrt(d_k)  # Q·Kᵀ/√d_k
    weights = np.exp(scores) / np.sum(np.exp(scores), axis=-1, keepdims=True)
    return weights @ V  # weighted sum of V

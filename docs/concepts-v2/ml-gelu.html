f(x) = x · Φ(x) ≈ 0.5x(1 + tanh(√(2/π)(x + 0.044715x³)))
# f(x) = x · Φ(x)  where Φ = standard normal CDF
# ≈ 0.5x(1 + tanh(√(2/π)(x + 0.044715x³)))
import numpy as np

def gelu(x):
    return 0.5 * x * (1 + np.tanh(
        np.sqrt(2/np.pi) * (x + 0.044715 * x**3)
    ))

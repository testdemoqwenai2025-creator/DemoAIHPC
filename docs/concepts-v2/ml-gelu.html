f(x) = ½x · (1 + tanh(√(2/π) · (x + 0.044715·x³)))
# f(x) = 0.5·x·(1 + tanh(√(2/π)·(x + 0.044715·x³)))
# tanh → np.tanh()   √ → np.sqrt()   π → np.pi
import numpy as np

def gelu(x):
    return 0.5 * x * (1 + np.tanh(np.sqrt(2 / np.pi) * (x + 0.044715 * x ** 3)))

def gelu_grad(x):
    # GELU derivative (closed form of the tanh approximation)
    k = np.sqrt(2 / np.pi)
    inner = k * (x + 0.044715 * x ** 3)
    t = np.tanh(inner)
    d_inner = k * (1 + 3 * 0.044715 * x ** 2)
    return 0.5 * (1 + t) + 0.5 * x * (1 - t ** 2) * d_inner

# Example
x = np.array([-2.0, -0.5, 0.0, 1.0, 3.0])
print(gelu(x))  # [-0.09 -0.085  0.    0.84  2.996]

y = γ · (x − μ_g) / √(σ²_g + ε) + β (per group of channels)
# y = γ · (x − μ_g) / √(σ²_g + ε) + β
# Split C channels into G groups, normalise per group per sample
# Useful when batch size is small (object detection, segmentation)
import numpy as np

def groupnorm(x, gamma, beta, num_groups=4, eps=1e-5):
    # x shape: (batch, channels, height, width)
    N, C, H, W = x.shape
    G = num_groups
    x_grouped = x.reshape(N, G, C // G, H, W)
    mu = x_grouped.mean(axis=(2, 3, 4), keepdims=True)
    var = x_grouped.var(axis=(2, 3, 4), keepdims=True)
    x_hat = (x_grouped - mu) / np.sqrt(var + eps)
    x_hat = x_hat.reshape(N, C, H, W)
    return gamma.reshape(1, C, 1, 1) * x_hat + beta.reshape(1, C, 1, 1)

# Example
x = np.random.randn(2, 8, 4, 4)  # 2 samples, 8 channels, 4×4 spatial
gamma = np.ones(8); beta = np.zeros(8)
print(groupnorm(x, gamma, beta, num_groups=4).shape)  # (2, 8, 4, 4)

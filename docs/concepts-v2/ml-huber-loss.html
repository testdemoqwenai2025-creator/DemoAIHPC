L = { ½e² if |e| ≤ δ, δ(|e| − ½δ) otherwise }
# L = ½e² if |e| ≤ δ, else δ(|e| − ½δ)
# δ → delta (threshold, typically 1.0)
import numpy as np

def huber(y_true, y_pred, delta=1.0):
    e = np.abs(y_true - y_pred)
    return np.mean(np.where(e <= delta,
                           0.5 * e ** 2,
                           delta * (e - 0.5 * delta)))

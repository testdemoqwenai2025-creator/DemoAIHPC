mₜ = β₁mₜ₋₁ + (1−β₁)gₜ vₜ = β₂vₜ₋₁ + (1−β₂)gₜ² θ ← θ − η·m̂/(√v̂+ε)
# m_t = β₁m_{t-1} + (1−β₁)g_t    (1st moment)
# v_t = β₂v_{t-1} + (1−β₂)g_t²   (2nd moment)
# θ ← θ − η · m̂/(√v̂ + ε)        (bias-corrected)
def adam_step(params, grads, m, v, t,
               lr=0.001, b1=0.9, b2=0.999, eps=1e-8):
    for i in range(len(params)):
        m[i] = b1*m[i] + (1-b1)*grads[i]
        v[i] = b2*v[i] + (1-b2)*grads[i]**2
        m_hat = m[i] / (1 - b1**t)
        v_hat = v[i] / (1 - b2**t)
        params[i] -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return params, m, v

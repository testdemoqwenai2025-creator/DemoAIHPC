D_KL(p‖q) = Σᵢ pᵢ · log(pᵢ/qᵢ)
# D_KL(p‖q) = Σ p · log(p/q)
# p → y_true (normalized)  q → y_pred (normalized)
import numpy as np

def kl_div(y_true, y_pred):
    p = np.abs(y_true) / np.sum(np.abs(y_true))
    q = np.abs(y_pred) / np.sum(np.abs(y_pred))
    return np.sum(p * np.log(p / (q + 1e-12)))

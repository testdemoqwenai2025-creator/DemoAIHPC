y = (x · m) / (1 − p) where mᵢ ~ Bernoulli(1 − p)
# y = (x · m) / (1 − p)   (training only)
# m_i ~ Bernoulli(1 − p)  (kept with prob 1-p, else 0)
# p → drop probability (e.g. 0.5)
# /(1-p) → keep expectation unchanged (inverted dropout)
# At inference: m = 1 (no scaling needed)
import numpy as np

def dropout(x, p=0.5, training=True, seed=None):
    if not training or p == 0:
        return x
    rng = np.random.default_rng(seed)
    mask = rng.binomial(1, 1 - p, size=x.shape)  # Bernoulli(1-p)
    return x * mask / (1 - p)  # inverted dropout

# Example: array of ones, 50% dropout
x = np.ones(10)
print(dropout(x, p=0.5, training=True, seed=42))
# values are 0 or 2 (kept: scaled by 1/(1-0.5)=2)
print(dropout(x, p=0.5, training=False))  # all 1s at inference
